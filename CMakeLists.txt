cmake_minimum_required(VERSION 3.20)
project(Cinderline LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CINDERLINE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CINDERLINE_BUILD_NATIVE "Build the macOS native playtest client" ON)

file(GLOB CINDERLINE_SIM_SOURCES CONFIGURE_DEPENDS Source/Cinderline/Private/Sim/*.cpp)
add_library(CinderlineSimulation STATIC ${CINDERLINE_SIM_SOURCES})
target_include_directories(CinderlineSimulation PUBLIC Source/Cinderline/Public)
if(MSVC)
  target_compile_options(CinderlineSimulation PRIVATE /W4)
else()
  target_compile_options(CinderlineSimulation PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CINDERLINE_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(CinderlineSimulation PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(CinderlineSimulation PUBLIC -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(CinderlineTests Tests/SimulationTests.cpp)
  target_link_libraries(CinderlineTests PRIVATE CinderlineSimulation)
  if(MSVC)
    target_compile_options(CinderlineTests PRIVATE /W4)
  else()
    target_compile_options(CinderlineTests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME simulation_rules COMMAND CinderlineTests)
  set_tests_properties(simulation_rules PROPERTIES TIMEOUT 120)
endif()

if(APPLE AND CINDERLINE_BUILD_NATIVE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  file(GLOB CINDERLINE_NATIVE_SOURCES CONFIGURE_DEPENDS Tools/Native/*.mm)
  add_executable(CinderlineNative ${CINDERLINE_NATIVE_SOURCES})
  target_compile_options(CinderlineNative PRIVATE -fobjc-arc)
  target_link_libraries(CinderlineNative PRIVATE CinderlineSimulation "-framework Cocoa" "-framework QuartzCore")
  if(BUILD_TESTING)
    add_test(NAME native_smoke COMMAND CinderlineNative --smoke-test)
    set_tests_properties(native_smoke PROPERTIES TIMEOUT 15)
    add_test(NAME native_render_stress COMMAND CinderlineNative --render-stress 3000)
    set_tests_properties(native_render_stress PROPERTIES TIMEOUT 120 LABELS "native;render")
  endif()
endif()
