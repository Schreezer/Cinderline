cmake_minimum_required(VERSION 3.20)
project(Cinderline LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CINDERLINE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CINDERLINE_BUILD_NATIVE "Build the macOS native playtest client" ON)
option(CINDERLINE_BUILD_SERVER "Build the authoritative online match worker" ON)

file(GLOB CINDERLINE_SIM_SOURCES CONFIGURE_DEPENDS Source/Cinderline/Private/Sim/*.cpp)
add_library(CinderlineSimulation STATIC ${CINDERLINE_SIM_SOURCES})
target_include_directories(CinderlineSimulation PUBLIC Source/Cinderline/Public)
if(CINDERLINE_BUILD_SERVER)
  add_executable(CinderlineMatchWorker Server/MatchWorker.cpp)
  target_link_libraries(CinderlineMatchWorker PRIVATE CinderlineSimulation)
endif()
if(MSVC)
  target_compile_options(CinderlineSimulation PRIVATE /W4)
else()
  target_compile_options(CinderlineSimulation PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(CINDERLINE_SANITIZE AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(CinderlineSimulation PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(CinderlineSimulation PUBLIC -fsanitize=address,undefined)
endif()

include(CTest)
if(BUILD_TESTING)
  # Keep compiler policy, linkage and CTest registration consistent as suites grow.
  function(cinderline_add_sim_test target source test_name timeout)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE CinderlineSimulation)
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME ${test_name} COMMAND ${target})
    set_tests_properties(${test_name} PROPERTIES TIMEOUT ${timeout})
    if(ARGN)
      set_tests_properties(${test_name} PROPERTIES LABELS "${ARGN}")
    endif()
  endfunction()

  cinderline_add_sim_test(CinderlineTests Tests/SimulationTests.cpp simulation_rules 120)
  cinderline_add_sim_test(CinderlineNavigationTests Tests/NavigationTests.cpp navigation_regressions 180 navigation)
  cinderline_add_sim_test(CinderlineNetworkTests Tests/NetworkTests.cpp network_protocol 120)
  cinderline_add_sim_test(CinderlineDefenseTests Tests/DefenseTests.cpp defense_regressions 120 defense)
  cinderline_add_sim_test(CinderlineAIDifficultyTests Tests/AIDifficultyTests.cpp ai_difficulty_regressions 120 ai)
  cinderline_add_sim_test(CinderlineProductionAllocationTests Tests/ProductionAllocationTests.cpp production_allocation_regressions 120 production network)
  cinderline_add_sim_test(CinderlineMatchLengthTests Tests/MatchLengthTests.cpp match_length_regressions 180 match-length navigation)
  cinderline_add_sim_test(CinderlineFourPlayerTests Tests/FourPlayerTests.cpp four_player_regressions 180 four-player navigation network)
endif()

if(APPLE AND CINDERLINE_BUILD_NATIVE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  file(GLOB CINDERLINE_NATIVE_SOURCES CONFIGURE_DEPENDS Tools/Native/*.mm)
  add_executable(CinderlineNative ${CINDERLINE_NATIVE_SOURCES})
  target_compile_options(CinderlineNative PRIVATE -fobjc-arc)
  target_link_libraries(CinderlineNative PRIVATE CinderlineSimulation "-framework Cocoa" "-framework QuartzCore")
  if(BUILD_TESTING)
    add_test(NAME native_smoke COMMAND CinderlineNative --smoke-test)
    set_tests_properties(native_smoke PROPERTIES TIMEOUT 15)
    add_test(NAME native_render_stress COMMAND CinderlineNative --render-stress 3000)
    set_tests_properties(native_render_stress PROPERTIES TIMEOUT 120 LABELS "native;render")
  endif()
endif()
